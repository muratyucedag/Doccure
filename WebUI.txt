function renderAutocomplete(q){
  const query = q.trim().toLowerCase();
  if(!query){ acBox.classList.remove('open'); acBox.innerHTML=''; return; }

  const matches = SUGGESTIONS.filter(s =>
    s.label.toLowerCase().includes(query) || s.sub.toLowerCase().includes(query) || s.type.includes(query));

  if(!matches.length){
    acBox.innerHTML = `<div class="ac-empty">No matches for “${q}”. Try a doctor, specialty, branch or service.</div>`;
    acBox.classList.add('open'); return;
  }

  /* group by type for a search-engine feel */
  const order = ['doctor','specialty','branch','hospital','service'];
  const labels = {doctor:'Doctors',specialty:'Specialties',branch:'Branches',hospital:'Hospitals',service:'Services'};
  let html = '';
  order.forEach(t=>{
    const grp = matches.filter(m=>m.type===t);
    if(!grp.length) return;
    html += `<div class="ac-group-label">${labels[t]}</div>`;
    grp.forEach(m=>{
      html += `<div class="ac-item" onclick="applyTerm('${m.label.replace(/'/g,"\\'")}')">
        <span class="ac-ico ${TYPE_CLASS[m.type]}"><i data-feather="${m.icon}" style="width:16px;height:16px"></i></span>
        <div><div>${highlight(m.label,q)}</div><div class="ac-sub">${m.sub}</div></div>
        <span class="ac-meta">${m.meta}</span>
      </div>`;
    });
  });
  acBox.innerHTML = html;
  acBox.classList.add('open');
  feather.replace();
  acIndex = -1;
}

function highlight(text,q){
  const i = text.toLowerCase().indexOf(q.toLowerCase());
  if(i<0) return text;
  return text.slice(0,i)+'<b style="color:#0aa5d1">'+text.slice(i,i+q.length)+'</b>'+text.slice(i+q.length);
}

  else if(e.key==='ArrowUp'){ e.preventDefault(); acIndex=Math.max(acIndex-1,0); }
  else if(e.key==='Enter'){ if(acIndex>=0&&items[acIndex]){ items[acIndex].click(); } else { runSearch(); } return; }
  else return;
  items.forEach((it,i)=>it.classList.toggle('active',i===acIndex));
});
